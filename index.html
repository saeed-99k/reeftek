كود تهيئة جدول الشحنات في Supabase (SQL Editor): 
create table if not exists public.shipments (
  id text primary key,
  shop_name text not null,
  city text not null,
  customer_name text,
  phone text not null,
  carrier text not null,
  boxes integer default 1,
  payment_status text default 'مدفوع',
  cod_amount text default '',
  invoice_no text default '',
  notes text default '',
  warnings jsonb default '[]'::jsonb,
  status text default 'قيد التجهيز',
  created_at timestamptz default now()
);
alter table public.shipments enable row level security;
drop policy if exists "Public shipments access" on public.shipments;
create policy "Public shipments access" on public.shipments for all to anon, authenticated using (true) with check (true);
